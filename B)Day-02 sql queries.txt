CREATE TABLE DEPT(
DID INT AUTO_INCREMENT PRIMARY KEY,
DNAME VARCHAR(30) UNIQUE NOT NULL,
DEMAIL VARCHAR(50) UNIQUE,
DPHONE BIGINT UNIQUE NOT NULL CHECK(LENGTH(DPHONE)=10),
DHOD VARCHAR(30) ,
PRINCIPAL VARCHAR(20) DEFAULT "SHIVUKUMAR");

CREATE DATABASE COMPANY;

USE COMPANY;

CREATE TABLE DEPAT(
DEPAT_ID INT AUTO_INCREMENT PRIMARY KEY,
DEPAT_CODE CHAR(5) NOT NULL UNIQUE,
DEPAT_NAME VARCHAR(50) DEFAULT 'BANGALORE',
BUDGET DECIMAL(12,2) DEFAULT 15000.00 CHECK(BUDGET >=0)
);


CREATE TABLE EMPLOYEE (
EMP_ID INT AUTO_INCREMENT PRIMARY KEY,
EMP_CODE CHAR(6) NOT NULL UNIQUE,
EMP_NAME VARCHAR(100) UNIQUE,
PHONE BIGINT UNIQUE,
SALARY DECIMAL(10,2) NOT NULL CHECK(SALARY >= 10000),
GENDER ENUM('MALE', 'FEMALE', 'OTHER'),
DOB DATE
);

CREATE TABLE EMPLOYEES (
EMP_ID INT AUTO_INCREMENT PRIMARY KEY,
EMP_CODE CHAR(6) NOT NULL UNIQUE,
EMP_NAME VARCHAR(100) UNIQUE,
PHONE BIGINT UNIQUE,
SALARY DECIMAL(10,2) NOT NULL CHECK(SALARY >= 10000),
GENDER ENUM('MALE', 'FEMALE', 'OTHER'),
DOB DATE,
DEPAT_ID INT NOT NULL,
FOREIGN KEY (DEPAT_ID) REFERENCES DEPAT(DEPAT_ID)
);


CREATE TABLE PROJECT(
PROJECT_ID INT AUTO_INCREMENT PRIMARY KEY,
PROJECT_CODE CHAR(6) NOT NULL UNIQUE,

PROJECT_GUIDE VARCHAR(30) DEFAULT "SATHISH",

START_TIME TIME DEFAULT '09:15:00'
);
